cmake_minimum_required(VERSION 3.10)
project(Game2048Engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(engine2048 src/core/Tile.cpp src/core/Board.cpp src/core/Game.cpp
    src/core/GameModes.cpp src/core/ScoreManager.cpp src/core/PlayerSystem.cpp
    src/core/HighScore.cpp src/core/Statistics.cpp src/core/SaveGame.cpp
    src/core/LoadGame.cpp src/core/FileManager.cpp src/core/Session.cpp
    src/core/DynamicBoard.cpp src/core/themeManager.cpp)
target_include_directories(engine2048 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
include(CTest)
if(BUILD_TESTING)
    add_executable(engine_regression tests/Regression.cpp)
    target_link_libraries(engine_regression PRIVATE engine2048)
    add_test(NAME backend_regression COMMAND engine_regression)
    add_executable(integration_regression tests/IntegrationRegression.cpp)
    target_link_libraries(integration_regression PRIVATE engine2048)
    add_test(NAME integration_regression COMMAND integration_regression)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/integration-persistence")
    add_test(NAME persistence_write COMMAND integration_regression write)
    add_test(NAME persistence_read COMMAND integration_regression read)
    set_tests_properties(persistence_write persistence_read PROPERTIES
        WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/integration-persistence")
    set_tests_properties(persistence_write PROPERTIES FIXTURES_SETUP saved_game)
    set_tests_properties(persistence_read PROPERTIES FIXTURES_REQUIRED saved_game)
endif()
option(GAME2048_BUILD_GUI "Build SFML desktop application" ON)
if(GAME2048_BUILD_GUI)
    find_package(SFML 2.6 CONFIG REQUIRED COMPONENTS graphics window system)
    add_library(ui2048 src/ui/Renderer.cpp src/ui/InputHandler.cpp src/ui/Application.cpp)
    target_link_libraries(ui2048 PUBLIC engine2048 sfml-graphics sfml-window sfml-system)
    add_executable(game2048 main.cpp)
    target_link_libraries(game2048 PRIVATE ui2048)
    if(BUILD_TESTING)
        add_executable(input_regression tests/InputRegression.cpp)
        target_link_libraries(input_regression PRIVATE ui2048)
        add_test(NAME input_regression COMMAND input_regression)
        add_executable(application_regression tests/ApplicationRegression.cpp)
        target_link_libraries(application_regression PRIVATE ui2048)
        add_test(NAME application_regression COMMAND application_regression)
    endif()
    add_custom_command(TARGET game2048 POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:game2048>/assets")
endif()
